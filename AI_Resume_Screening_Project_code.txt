<script>
    function sortResumes() {
        const keywordInput = document.getElementById('keywords').value;
        const fileInput = document.getElementById('resumeFiles');
        const table = document.getElementById('resultsTable');
        const tbody = document.getElementById('resultsBody');

        
        tbody.innerHTML = "";

        
        if (!keywordInput.trim()) {
            alert("Please enter job keywords first.");
            return;
        }
        const keywords = keywordInput.toLowerCase().split(',').map(kw => kw.trim()).filter(kw => kw.length > 0);

        const files = fileInput.files;
        if (files.length === 0) {
            alert("Please upload at least one .txt resume file.");
            return;
        }

        let processedCount = 0;
        const candidatesData = [];

        
        for (let i = 0; i < files.length; i++) {
            const file = files[i];
            const reader = new FileReader();

            reader.onload = function(e) {
                const text = e.target.result.toLowerCase();
                let matchCount = 0;
                let matchedWords = [];

                
                keywords.forEach(keyword => {
                    if (text.includes(keyword)) {
                        matchCount++;
                        matchedWords.push(keyword);
                    }
                });

                
                const matchPercentage = Math.round((matchCount / keywords.length) * 100);

                candidatesData.push({
                    name: file.name.replace('.txt', ''),
                    score: matchPercentage,
                    matches: matchedWords
                });

                processedCount++;

                
                if (processedCount === files.length) {
                    displayResults(candidatesData, table, tbody);
                }
            };

            reader.readAsText(file);
        }
    }

    function displayResults(data, table, tbody) {
        
        data.sort((a, b) => b.score - a.score);

        
        data.forEach((candidate, index) => {
            const row = tbody.insertRow();
            
            const cellRank = row.insertCell(0);
            const cellName = row.insertCell(1);
            const cellScore = row.insertCell(2);
            const cellMatches = row.insertCell(3);

            cellRank.innerHTML = `<strong>#${index + 1}</strong>`;
            cellName.textContent = candidate.name;
            cellScore.innerHTML = `<span class="badge ${candidate.score > 50 ? 'matched' : ''}">${candidate.score}%</span>`;
            cellMatches.textContent = candidate.matches.join(', ') || 'None';
        });

        
        table.style.display = 'table';
    }
</script>
